# -------------------------------------------------------
# AI DEMAND PREDICTION
# -------------------------------------------------------

st.markdown("""
<div class="section-title">
🤖 AI Regional Demand Intelligence
</div>
""", unsafe_allow_html=True)

X = df[
    [
        'Emergency_Requests',
        'Daily_Usage'
    ]
]

y = df['Units_Available']

model = LinearRegression()

model.fit(X, y)

city_data = df.groupby(
    ['City', 'Blood_Group']
).agg({
    'Emergency_Requests':'sum',
    'Daily_Usage':'sum'
}).reset_index()

predictions = []

for index, row in city_data.iterrows():

    pred = model.predict([[
        row['Emergency_Requests'],
        row['Daily_Usage']
    ]])[0]

    predictions.append(pred)

city_data['Predicted_Demand'] = predictions

city_data = city_data.sort_values(
    by='Predicted_Demand',
    ascending=False
)

for index, row in city_data.head(8).iterrows():

    if row['Predicted_Demand'] > 80:
        badge = "badge-high"
        level = "CRITICAL DEMAND"

    elif row['Predicted_Demand'] > 45:
        badge = "badge-medium"
        level = "HIGH DEMAND"

    else:
        badge = "badge-low"
        level = "MODERATE DEMAND"

    st.markdown(f"""
    <div class="ai-card">

    <h2>📍 {row['City']}</h2>

    🩸 Blood Group:
    <h3>{row['Blood_Group']}</h3>

    <span class="{badge}">
    {level}
    </span>

    <br><br>

    📈 Predicted Required Units:
    <h1>{row['Predicted_Demand']:.2f}</h1>

    🧠 AI Analysis:
    This region is expected to require
    additional <b>{row['Blood_Group']}</b>
    blood units due to increasing emergency
    requests and daily consumption trends.

    </div>
    """, unsafe_allow_html=True)
